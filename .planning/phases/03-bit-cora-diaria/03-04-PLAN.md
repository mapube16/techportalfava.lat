  ['America/Bogota', 300], ['Europe/Rome', -120],
  ['America/Sao_Paulo', 180], ['Pacific/Kiritimati', -840],
];

describe.each(HUSOS)('servidor en %s', (zona, offsetEsperado) => {
  beforeAll(() => { process.env.TZ = zona; });
  afterAll(() => { process.env.TZ = 'America/Bogota'; });
